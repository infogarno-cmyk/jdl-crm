// supabase/functions/creo-ai/index.ts // Геннадий для раздела «Крео»: // kind "tag" — по кадрам и сценарию определяет стиль, тип хука, сценарий, доп. теги и строку хука // kind "script" — пишет сценарий крео по параметрам (и по образцу MWP) // kind "advice" — советы по пачкам: что работает, новые хуки, визуал, что не повторять // Ключ: секрет ANTHROPIC_API_KEY (Project Settings → Edge Functions → Secrets). // Модель можно поменять секретом CREO_AI_MODEL. const CORS = { "Access-Control-Allow-Origin": "*", "Access-Control-Allow-Headers": "authorization, x-client-info, apikey, content-type", "Access-Control-Allow-Methods": "POST, OPTIONS", } const json = (d: unknown, status = 200) => new Response(JSON.stringify(d), { status, headers: { ...CORS, "Content-Type": "application/json" } }) const MODEL = Deno.env.get("CREO_AI_MODEL") || "claude-sonnet-5-5" const KEY = Deno.env.get("ANTHROPIC_API_KEY") || "" async function ask(system: string, content: any[], maxTokens = 1500, _temperature = 0.7): Promise { const r = await fetch("https://api.anthropic.com/v1/messages", { method: "POST", headers: { "x-api-key": KEY, "anthropic-version": "2023-06-01", "content-type": "application/json" }, body: JSON.stringify({ model: MODEL, max_tokens: maxTokens, system, messages: [{ role: "user", content }] }), }) const j = await r.json() if (j.error) throw new Error(j.error.message || "anthropic error") return (j.content || []).filter((c: any) => c.type === "text").map((c: any) => c.text).join("\n") } function pickJson(t: string) { const a = t.indexOf("{"), b = t.lastIndexOf("}") if (a < 0 || b < a) throw new Error("ответ без JSON") return JSON.parse(t.slice(a, b + 1)) } function imgBlocks(frames: string[], max = 4) { return (frames || []).slice(0, max).map((u) => { const m = String(u).match(/^data:(image\/[a-z]+);base64,(.+)$/) return m ? { type: "image", source: { type: "base64", media_type: m[1], data: m[2] } } : null }).filter(Boolean) } const SYS = `Ты Геннадий — помощник медиабаеров GARNO Custom Furniture (мебель на заказ: кухни, шкафы, гардеробные; Варшава и Прага). " — " + desc[k][v] : ""}`).join("\n") // старый формат: одно крео const items = Array.isArray(p.items) ? p.items : [{ key: "0", name: p.name, script: p.script, frames: p.frames }] const ex = (p.examples || []).slice(0, 30).map((e: any) => `«${e.hook_line || e.name}» → стиль ${e.style}, хук ${e.hook}, сценарий ${e.scen}`).join("\n") const fx = (p.corrections || []).slice(0, 25).map((e: any) => `«${e.hook_line || e.name}»: ты поставил ${e.was.style}/${e.was.hook}/${e.was.scen}, баер исправил на ${e.now.style}/${e.now.hook}/${e.now.scen}`).join("\n") const rules = `Определи теги каждого крео. Структура: 1) первая строка — хук из крео, с цифрой (цена/срок/размер) если есть; 2) 2–3 коротких абзаца выгод, без воды; 3) призыв рассчитать цену в калькуляторе; 4) 3–5 хэштегов на том же языке.
